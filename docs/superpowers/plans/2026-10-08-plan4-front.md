> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 챗봇 동작을 확인하는 내부 도구를 만든다 — 테스트 채팅 + 실시간 판단 패널, 대화 로그 + trace 서랍, 검수 대기, 지식 데이터, 설정.

**Tech Stack:** Vite ^8.3, React ^19.3, TypeScript ~6.0.2, Tailwind ^4.3 (`@tailwindcss/vite`), shadcn CLI 4.21.4, `@tanstack/react-query` ^5.104, `socket.io-client` 4.8.3, `sonner` ^2, `lucide-react`, Vitest ^5 + jsdom + Testing Library

## Global Constraints

- 공통 규칙(한국어, 커밋 trailer, push 금지, `.env*` 읽기 금지)을 따른다.
- **토큰·키를 번들에 넣지 않는다.** `VITE_` 변수는 `VITE_API_URL`(서버 주소, 비밀 아님)만 쓴다. 접속 토큰은 사용자가 설정 화면에서 입력하고 **`sessionStorage`에만** 저장하며 로그아웃 시 즉시 삭제한다.
- 서버가 준 텍스트(답변·원문·trace)는 **일반 텍스트로만 렌더링**한다. `dangerouslySetInnerHTML` 금지, Markdown 해석 금지. 줄바꿈은 CSS `white-space: pre-wrap`.
- 소켓 이벤트는 ack보다 먼저 올 수 있다 → 턴 상태는 `clientMsgId` 기준 upsert. 재연결 시 `session:start`로 최신 50턴을 받아 upsert한다.
- "다시 보내기"는 **새 clientMsgId** + `retryOfTurnSeq`로 보낸다.
- 서버 미들웨어가 연결을 거부하면(`connect_error`) socket.io-client는 자동 재연결하지 않는다 → 오류 코드를 보여주고 사용자가 다시 연결한다.
- 최근 메이저가 바뀐 라우터·테이블·모킹 라이브러리는 쓰지 않는다(화면 전환은 상태, 표는 shadcn Table 마크업, 테스트 모킹은 직접 만든 가짜 객체).

`jev-front/src/lib/token-store.ts`:
```ts
const KEY = "jev-chat.token";

/** 접속 토큰은 탭 수명(sessionStorage)으로만 보관한다. 번들·localStorage에 두지 않는다. */
export function getToken(): string | null {
  return sessionStorage.getItem(KEY);
}
export function setToken(token: string): void {
  const t = token.trim();
  if (t) sessionStorage.setItem(KEY, t);
}
export function clearToken(): void {
  sessionStorage.removeItem(KEY);
}

export const API_URL: string = import.meta.env.VITE_API_URL ?? "http://localhost:3000";
```

**Interfaces:**
- Produces:
  - `class ApiRequestError extends Error { status: number; code: ErrorCode; retryAfterMs?: number }`
  - `class ApiClient` — `constructor(baseUrl: string, getToken: () => string | null, fetchImpl?: typeof fetch)`, `get<T>(path: string, schema: z.ZodType<T>): Promise<T>`, `post<T>(path: string, body: unknown, schema: z.ZodType<T>): Promise<T>`
  - `type UiTurn = Turn & { localId: string; stage?: "queued" | "judging" | "answering"; sendError?: { code: ErrorCode; message: string; retryable: boolean } }`
  - `interface TurnsState { sessionId: string | null; byClientId: Record<string, UiTurn>; traces: Record<string, Record<string, unknown>> }`
  - `type TurnsAction = { type: "hydrate"; sessionId: string; turns: Turn[] } | { type: "localSend"; clientMsgId: string; text: string } | { type: "ack"; clientMsgId: string; turnId: string; turnSeq: number } | { type: "ackFailed"; clientMsgId: string; error: { code; message; retryable } } | { type: "status"; e: ChatStatusEvent } | { type: "done"; e: ChatDoneEvent } | { type: "error"; e: ChatErrorEvent } | { type: "trace"; e: ChatTraceEvent } | { type: "reset" }`
  - `turnsReducer(state, action): TurnsState`, `initialTurns: TurnsState`, `orderedTurns(state): UiTurn[]` (turnSeq 오름차순, 아직 seq 없는 로컬 전송은 맨 뒤 입력 순)

```ts
/** 실제 Response를 돌려주는 가짜 fetch. 호출 기록을 남긴다. */
export function fakeFetch(routes: Record<string, { status: number; body: unknown }>) {
  const calls: { url: string; init?: RequestInit }[] = [];
  const fn = async (url: string, init?: RequestInit) => {
    calls.push({ url, init });
    const key = `${init?.method ?? "GET"} ${new URL(url).pathname}${new URL(url).search}`;
    const r = routes[key] ?? { status: 404, body: { error: { code: "NOT_FOUND", message: "없음" } } };
    return new Response(JSON.stringify(r.body), { status: r.status, headers: { "content-type": "application/json" } });
  };
  return { fn: fn as unknown as typeof fetch, calls };
}

type Handler = (...args: unknown[]) => void;

/** socket.io-client Socket의 필요한 부분만 흉내 낸다. */
export class FakeSocket {
  connected = false;
  active = true;
  readonly emitted: { event: string; payload: unknown }[] = [];
  private handlers = new Map<string, Set<Handler>>();
  ackResponder: (event: string, payload: unknown) => unknown = () => ({ ok: true, data: {} });

  on(event: string, h: Handler) {
    if (!this.handlers.has(event)) this.handlers.set(event, new Set());
    this.handlers.get(event)!.add(h);
    return this;
  }
  off(event: string, h?: Handler) {
    if (h) this.handlers.get(event)?.delete(h);
    else this.handlers.delete(event);
    return this;
  }
  once(event: string, h: Handler) {
    const w: Handler = (...a) => {
      this.off(event, w);
      h(...a);
    };
    return this.on(event, w);
  }
  /** 서버 → 클라이언트 이벤트 흉내 */
  serverEmit(event: string, ...args: unknown[]) {
    for (const h of [...(this.handlers.get(event) ?? [])]) h(...args);
  }
  connect() {
    this.connected = true;
    this.serverEmit("connect");
    return this;
  }
  disconnect() {
    this.connected = false;
    this.serverEmit("disconnect", "io client disconnect");
    return this;
  }
  timeout(_ms: number) {
    return {
      emitWithAck: async (event: string, payload: unknown) => {
        this.emitted.push({ event, payload });
        return this.ackResponder(event, payload);
      },
    };
  }
}
```

`jev-front/src/lib/turns.spec.ts`:
```ts
import { describe, expect, it } from "vitest";
import { initialTurns, orderedTurns, turnsReducer, type TurnsAction } from "./turns";

const sid = "00000000-0000-4000-8000-000000000001";
const cid = (n: number) => `00000000-0000-4000-8000-${String(n).padStart(12, "0")}`;
const run = (actions: TurnsAction[]) => actions.reduce(turnsReducer, initialTurns);

describe("turnsReducer", () => {
  it("로컬 전송 → ack → status → done", () => {
    const s = run([
      { type: "hydrate", sessionId: sid, turns: [] },
      { type: "localSend", clientMsgId: cid(1), text: "법인카드 한도" },
      { type: "ack", clientMsgId: cid(1), turnId: "t1", turnSeq: 1 },
      { type: "status", e: { sessionId: sid, clientMsgId: cid(1), turnSeq: 1, stage: "judging" } },
      { type: "done", e: { sessionId: sid, clientMsgId: cid(1), turnSeq: 1, turnId: "t1", text: "50만 원", route: "faq", sources: [], traceId: "tr1" } },
    ]);
    expect(orderedTurns(s)).toEqual([expect.objectContaining({ turnSeq: 1, status: "completed", assistantText: "50만 원", route: "faq", traceId: "tr1" })]);
  });

  it("이벤트가 ack보다 먼저 와도 같은 턴으로 합친다", () => {
    const s = run([
      { type: "hydrate", sessionId: sid, turns: [] },
      { type: "localSend", clientMsgId: cid(1), text: "q" },
      { type: "done", e: { sessionId: sid, clientMsgId: cid(1), turnSeq: 1, turnId: "t1", text: "a", route: "faq", sources: [], traceId: "tr1" } },
      { type: "ack", clientMsgId: cid(1), turnId: "t1", turnSeq: 1 },
    ]);
    expect(orderedTurns(s)).toHaveLength(1);
    expect(orderedTurns(s)[0]).toMatchObject({ status: "completed", assistantText: "a" });
  });

  it("hydrate는 turnSeq 기준 upsert — 이미 본 턴의 늦은 완료도 반영", () => {
    const s = run([
      { type: "hydrate", sessionId: sid, turns: [{ turnId: "t1", turnSeq: 1, clientMsgId: cid(1), userText: "q", status: "processing" }] },
      { type: "hydrate", sessionId: sid, turns: [{ turnId: "t1", turnSeq: 1, clientMsgId: cid(1), userText: "q", status: "failed", error: { code: "RESTARTED", retryable: true } }] },
    ]);
    expect(orderedTurns(s)[0]).toMatchObject({ status: "failed", error: { code: "RESTARTED" } });
  });

  it("완료된 턴은 늦게 온 processing hydrate로 되돌아가지 않는다", () => {
    const s = run([
      { type: "hydrate", sessionId: sid, turns: [] },
      { type: "done", e: { sessionId: sid, clientMsgId: cid(1), turnSeq: 1, turnId: "t1", text: "a", route: "faq", sources: [], traceId: "tr1" } },
      { type: "hydrate", sessionId: sid, turns: [{ turnId: "t1", turnSeq: 1, clientMsgId: cid(1), userText: "q", status: "processing" }] },
    ]);
    expect(orderedTurns(s)[0]?.status).toBe("completed");
  });

  it("ack 실패는 sendError로 남고, 다른 세션 이벤트는 무시", () => {
```

```ts
import { ApiErrorSchema, type ErrorCode } from "@jev-chat/protocol";
import type { z } from "zod";

export class ApiRequestError extends Error {
  constructor(readonly status: number, readonly code: ErrorCode, message: string, readonly retryAfterMs?: number) {
    super(message);
    this.name = "ApiRequestError";
  }
}

export class ApiClient {
  constructor(
    private readonly baseUrl: string,
    private readonly getToken: () => string | null,
    private readonly fetchImpl: typeof fetch = (...a) => fetch(...a),
  ) {}

  get<T>(path: string, schema: z.ZodType<T>): Promise<T> {
    return this.request("GET", path, undefined, schema);
  }

  post<T>(path: string, body: unknown, schema: z.ZodType<T>): Promise<T> {
    return this.request("POST", path, body, schema);
  }

  private async request<T>(method: string, path: string, body: unknown, schema: z.ZodType<T>): Promise<T> {
    const token = this.getToken();
    if (!token) throw new ApiRequestError(401, "UNAUTHORIZED", "접속 토큰을 먼저 입력하세요.");
    const res = await this.fetchImpl(`${this.baseUrl}${path}`, {
      method,
      headers: { authorization: `Bearer ${token}`, ...(body !== undefined ? { "content-type": "application/json" } : {}) },
      ...(body !== undefined ? { body: JSON.stringify(body) } : {}),
    });
    const json: unknown = await res.json().catch(() => null);
    if (!res.ok) {
      const err = ApiErrorSchema.safeParse(json);
      if (err.success) throw new ApiRequestError(res.status, err.data.error.code, err.data.error.message, err.data.error.retryAfterMs);
      throw new ApiRequestError(res.status, "INTERNAL", `요청 실패 (${res.status})`);
    }
    const parsed = schema.safeParse(json);
    if (!parsed.success) throw new ApiRequestError(res.status, "INTERNAL", "서버 응답 형식이 올바르지 않습니다.");
    return parsed.data;
  }
}
```

```ts
import type { ChatDoneEvent, ChatErrorEvent, ChatStatusEvent, ChatTraceEvent, ErrorCode, Turn } from "@jev-chat/protocol";

export type UiTurn = Turn & {
  localId: string;
  stage?: "queued" | "judging" | "answering";
  sendError?: { code: ErrorCode; message: string; retryable: boolean };
};

export interface TurnsState {
  sessionId: string | null;
  byClientId: Record<string, UiTurn>;
  traces: Record<string, Record<string, unknown>>;
}

export type TurnsAction =
  | { type: "hydrate"; sessionId: string; turns: Turn[] }
  | { type: "localSend"; clientMsgId: string; text: string }
  | { type: "ack"; clientMsgId: string; turnId: string; turnSeq: number }
  | { type: "ackFailed"; clientMsgId: string; error: { code: ErrorCode; message: string; retryable: boolean } }
  | { type: "status"; e: ChatStatusEvent }
  | { type: "done"; e: ChatDoneEvent }
  | { type: "error"; e: ChatErrorEvent }
  | { type: "trace"; e: ChatTraceEvent }
  | { type: "reset" };

export const initialTurns: TurnsState = { sessionId: null, byClientId: {}, traces: {} };

const TERMINAL = new Set(["completed", "failed"]);
let localCounter = 0;

function base(clientMsgId: string, userText = ""): UiTurn {
  return { localId: `l${++localCounter}`, turnId: "", turnSeq: 0, clientMsgId, userText, status: "processing" };
}

function put(state: TurnsState, clientMsgId: string, patch: (t: UiTurn) => UiTurn): TurnsState {
  const prev = state.byClientId[clientMsgId] ?? base(clientMsgId);
  return { ...state, byClientId: { ...state.byClientId, [clientMsgId]: patch(prev) } };
}

export function turnsReducer(state: TurnsState, a: TurnsAction): TurnsState {
  switch (a.type) {
    case "reset":
      return initialTurns;
    case "hydrate": {
      let s: TurnsState = state.sessionId === a.sessionId ? state : { ...initialTurns, sessionId: a.sessionId };
      for (const t of a.turns) {
        s = put(s, t.clientMsgId, (prev) => (TERMINAL.has(prev.status) && !TERMINAL.has(t.status) ? prev : { ...prev, ...t, stage: undefined }));
      }
      return s;
    }
    case "localSend":
      return put(state, a.clientMsgId, (prev) => ({ ...prev, userText: a.text, status: "processing", stage: "queued", sendError: undefined }));
    case "ack":
      return put(state, a.clientMsgId, (prev) => ({ ...prev, turnId: a.turnId, turnSeq: a.turnSeq }));
    case "ackFailed":
      return put(state, a.clientMsgId, (prev) => ({ ...prev, sendError: a.error, stage: undefined }));
    case "status":
      if (a.e.sessionId !== state.sessionId) return state;
      return put(state, a.e.clientMsgId, (prev) => (TERMINAL.has(prev.status) ? prev : { ...prev, turnSeq: a.e.turnSeq, stage: a.e.stage }));
    case "done":
      if (a.e.sessionId !== state.sessionId) return state;
      return put(state, a.e.clientMsgId, (prev) => ({
        ...prev,
        turnId: a.e.turnId,
        turnSeq: a.e.turnSeq,
        status: "completed",
        assistantText: a.e.text,
        route: a.e.route,
        sources: a.e.sources,
        traceId: a.e.traceId,
        ...(a.e.error ? { error: a.e.error } : {}),
        stage: undefined,
      }));
    case "error":
      if (a.e.sessionId !== state.sessionId) return state;
      return put(state, a.e.clientMsgId, (prev) => ({ ...prev, turnSeq: a.e.turnSeq, status: "failed", error: { code: a.e.code, retryable: a.e.retryable }, stage: undefined }));
    case "trace":
      return { ...state, traces: { ...state.traces, [a.e.traceId]: a.e.trace } };
  }
}

export function orderedTurns(state: TurnsState): UiTurn[] {
  const all = Object.values(state.byClientId);
  const seq = all.filter((t) => t.turnSeq > 0).sort((a, b) => a.turnSeq - b.turnSeq);
  const pending = all.filter((t) => t.turnSeq === 0).sort((a, b) => Number(a.localId.slice(1)) - Number(b.localId.slice(1)));
  return [...seq, ...pending];
}
```

```ts
import { io } from "socket.io-client";
import {
  ErrorCodeSchema,
  MessageTextSchema,
  PROTOCOL_VERSION,
  type Ack,
  type ChatSendResponse,
  type SessionStartResponse,
} from "@jev-chat/protocol";
import type { TurnsAction } from "./turns";

export type ConnState =
  | { kind: "idle" }
  | { kind: "connecting" }
  | { kind: "connected" }
  | { kind: "disconnected"; reason: string }
  | { kind: "rejected"; code: string };

export interface SocketLike {
  on(event: string, h: (...a: never[]) => void): unknown;
  off(event: string, h?: (...a: never[]) => void): unknown;
  connect(): unknown;
  disconnect(): unknown;
  timeout(ms: number): { emitWithAck(event: string, payload: unknown): Promise<unknown> };
}

interface Opts {
  url: string;
  token: string;
  dispatch: (a: TurnsAction) => void;
  onConn: (s: ConnState) => void;
  ioFactory?: (url: string, opts: object) => SocketLike;
  newId?: () => string;
}

const ACK_TIMEOUT = 5000;

export class ChatClient {
  private socket: SocketLike | null = null;
  private currentSession: string | null = null;
  private readonly newId: () => string;

  constructor(private readonly opts: Opts) {
    this.newId = opts.newId ?? (() => crypto.randomUUID());
  }

  get sessionId(): string | null {
    return this.currentSession;
  }

  connect(): void {
    const factory = this.opts.ioFactory ?? ((url: string, o: object) => io(url, o) as unknown as SocketLike);
    const s = factory(`${this.opts.url}/chat`, {
      auth: { token: this.opts.token, protocolVersion: PROTOCOL_VERSION },
      transports: ["websocket"],
      reconnection: true,
      autoConnect: false,
    });
    this.socket = s;
    const d = this.opts.dispatch;
    s.on("connect", () => {
      this.opts.onConn({ kind: "connected" });
      void this.startSession(this.currentSession ?? undefined);
    });
    s.on("disconnect", (reason: string) => this.opts.onConn({ kind: "disconnected", reason }));
    s.on("connect_error", (err: Error) => {
      const code = ErrorCodeSchema.safeParse(err.message);
      this.opts.onConn(code.success ? { kind: "rejected", code: code.data } : { kind: "disconnected", reason: err.message });
    });
    s.on("chat:status", (e: never) => d({ type: "status", e }));
    s.on("chat:done", (e: never) => d({ type: "done", e }));
    s.on("chat:error", (e: never) => d({ type: "error", e }));
    s.on("chat:trace", (e: never) => d({ type: "trace", e }));
    this.opts.onConn({ kind: "connecting" });
    s.connect();
  }

  async startSession(sessionId?: string, beforeTurnSeq?: number): Promise<{ ok: boolean; hasMore?: boolean; nextBeforeTurnSeq?: number }> {
    if (!this.socket) return { ok: false };
    const payload = { ...(sessionId ? { sessionId } : {}), ...(beforeTurnSeq ? { beforeTurnSeq } : {}) };
    const res = (await this.socket.timeout(ACK_TIMEOUT).emitWithAck("session:start", payload).catch(() => null)) as Ack<SessionStartResponse> | null;
    if (!res || !res.ok) return { ok: false };
    this.currentSession = res.data.sessionId;
    this.opts.dispatch({ type: "hydrate", sessionId: res.data.sessionId, turns: res.data.turns });
    return { ok: true, hasMore: res.data.hasMore, ...(res.data.nextBeforeTurnSeq ? { nextBeforeTurnSeq: res.data.nextBeforeTurnSeq } : {}) };
  }

  async send(text: string, retryOfTurnSeq?: number): Promise<void> {
    const parsed = MessageTextSchema.safeParse(text);
    if (!parsed.success) throw new Error(parsed.error.issues[0]?.message ?? "메시지 형식 오류");
    if (!this.socket || !this.currentSession) throw new Error("연결되지 않았습니다.");
    const clientMsgId = this.newId();
    this.opts.dispatch({ type: "localSend", clientMsgId, text: parsed.data });
    try {
      const res = (await this.socket
        .timeout(ACK_TIMEOUT)
        .emitWithAck("chat:send", { sessionId: this.currentSession, clientMsgId, text: parsed.data, ...(retryOfTurnSeq ? { retryOfTurnSeq } : {}) })) as Ack<ChatSendResponse>;
      if (res.ok) this.opts.dispatch({ type: "ack", clientMsgId, turnId: res.data.turnId, turnSeq: res.data.turnSeq });
      else this.opts.dispatch({ type: "ackFailed", clientMsgId, error: res.error });
    } catch {
      this.opts.dispatch({ type: "ackFailed", clientMsgId, error: { code: "INTERNAL", message: "응답이 없습니다. 다시 시도해주세요.", retryable: true } });
    }
  }

  close(): void {
    this.socket?.disconnect();
    this.socket = null;
  }
}
```

```ts
import { useCallback, useEffect, useMemo, useReducer, useRef, useState } from "react";
import { ChatClient, type ConnState } from "@/lib/chat-client";
import { API_URL } from "@/lib/token-store";
import { initialTurns, orderedTurns, turnsReducer, type UiTurn } from "@/lib/turns";

export function useChat(token: string | null) {
  const [state, dispatch] = useReducer(turnsReducer, initialTurns);
  const [conn, setConn] = useState<ConnState>({ kind: "idle" });
  const [older, setOlder] = useState<number | null>(null);
  const clientRef = useRef<ChatClient | null>(null);

  useEffect(() => {
    if (!token) return;
    const c = new ChatClient({ url: API_URL, token, dispatch, onConn: setConn });
    clientRef.current = c;
    c.connect();
    return () => c.close();
  }, [token]);

  const send = useCallback((text: string) => clientRef.current?.send(text) ?? Promise.resolve(), []);
  const retry = useCallback((t: UiTurn) => clientRef.current?.send(t.userText, t.turnSeq || undefined) ?? Promise.resolve(), []);
  const newSession = useCallback(async () => {
    dispatch({ type: "reset" });
    const r = await clientRef.current?.startSession();
    setOlder(r?.nextBeforeTurnSeq ?? null);
  }, []);
  const loadOlder = useCallback(async () => {
    const c = clientRef.current;
    if (!c?.sessionId || older === null) return;
    const r = await c.startSession(c.sessionId, older);
    setOlder(r.hasMore && r.nextBeforeTurnSeq ? r.nextBeforeTurnSeq : null);
  }, [older]);

  const turns = useMemo(() => orderedTurns(state), [state]);
  return { state, turns, conn, send, retry, newSession, loadOlder, hasOlder: older !== null };
}
```

```ts
import { z } from "zod";

const Num = z.number().finite();
const Candidate = z.object({
  kind: z.enum(["faq", "chunk"]),
  id: z.string(),
  bm25Rank: Num,
  relevance: Num.optional(),
  faqProb: Num.optional(),
  status: z.string().optional(),
});
const TraceSchema = z.object({
  route: z.string(),
  bStatus: z.string(),
  intent: z.string().nullable().optional(),
  intentProbs: z.record(z.string(), Num).nullable().optional(),
  inScope: Num.nullable().optional(),
  ambiguity: Num.nullable().optional(),
  faqChoice: z.string().nullable().optional(),
  faqProb: Num.nullable().optional(),
  totalInputTokens: Num.optional(),
  latencyMs: z.object({ retrieval: Num, turn: Num.nullable(), relevance: Num.nullable(), total: Num }).optional(),
  candidates: z.array(Candidate).default([]),
});

export interface TraceView {
  route: string;
  bStatus: string;
  intent: string | null;
  intents: { id: string; prob: number }[];
  inScope: number | null;
  ambiguity: number | null;
  faqChoice: string | null;
  faqCandidates: { id: string; rank: number; prob: number | null }[];
  chunkCandidates: { id: string; rank: number; relevance: number | null; status: string }[];
  latency: { retrieval: number; turn: number | null; relevance: number | null; total: number } | null;
  totalInputTokens: number | null;
}

export function toTraceView(raw: unknown): TraceView | null {
  const p = TraceSchema.safeParse(raw);
  if (!p.success) return null;
  const t = p.data;
  return {
    route: t.route,
    bStatus: t.bStatus,
    intent: t.intent ?? null,
    intents: Object.entries(t.intentProbs ?? {}).map(([id, prob]) => ({ id, prob })).sort((a, b) => b.prob - a.prob),
    inScope: t.inScope ?? null,
    ambiguity: t.ambiguity ?? null,
    faqChoice: t.faqChoice ?? null,
    faqCandidates: t.candidates.filter((c) => c.kind === "faq").map((c) => ({ id: c.id, rank: c.bm25Rank, prob: c.faqProb ?? null })),
    chunkCandidates: t.candidates.filter((c) => c.kind === "chunk").map((c) => ({ id: c.id, rank: c.bm25Rank, relevance: c.relevance ?? null, status: c.status ?? "-" })),
    latency: t.latencyMs ?? null,
    totalInputTokens: t.totalInputTokens ?? null,
  };
}
```

```tsx
import { useState } from "react";
import { Button } from "@/components/ui/button";
import { Textarea } from "@/components/ui/textarea";

export function Composer({ onSend, disabled }: { onSend: (text: string) => Promise<void>; disabled: boolean }) {
  const [text, setText] = useState("");
  const [error, setError] = useState<string | null>(null);
  const submit = async () => {
    try {
      setError(null);
      await onSend(text);
      setText("");
    } catch (e) {
      setError((e as Error).message);
    }
  };
  return (
    <div className="space-y-1">
      <div className="flex gap-2">
        <Textarea
          value={text}
          onChange={(e) => setText(e.target.value)}
          onKeyDown={(e) => {
            if (e.key === "Enter" && !e.shiftKey && !e.nativeEvent.isComposing) {
              e.preventDefault();
              void submit();
            }
          }}
          placeholder="HB-ERP 규정이나 사용법을 물어보세요 (Shift+Enter 줄바꿈)"
          disabled={disabled}
        />
        <Button onClick={() => void submit()} disabled={disabled || !text.trim()}>보내기</Button>
      </div>
      <div className="flex justify-between text-xs text-muted-foreground">
        <span className="text-destructive">{error}</span>
        <span>{[...text].length} / 1000</span>
      </div>
    </div>
  );
}
```

```ts
import { ApiClient, ApiRequestError } from "./api-client";
import { API_URL, getToken } from "./token-store";

export const api = new ApiClient(API_URL, getToken);

export function errorText(e: unknown): string {
  if (!(e instanceof ApiRequestError)) return "알 수 없는 오류";
  if (e.code === "UNAUTHORIZED") return "토큰을 확인하세요.";
  if (e.code === "FORBIDDEN") return "권한이 없습니다.";
  if (e.code === "RATE_LIMITED") return `요청이 많습니다. ${Math.ceil((e.retryAfterMs ?? 0) / 1000)}초 후 다시 시도하세요.`;
  return e.message;
}
```

```tsx
import { useState } from "react";
import type { ReviewRequest } from "@jev-chat/protocol";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Textarea } from "@/components/ui/textarea";

const VERDICTS = [
  { value: "correct", label: "맞음" },
  { value: "wrong", label: "틀림" },
  { value: "partial", label: "부분" },
] as const;

export function ReviewForm({ onSubmit }: { onSubmit: (r: ReviewRequest) => Promise<void> }) {
  const [verdict, setVerdict] = useState<ReviewRequest["verdict"] | null>(null);
  const [intent, setIntent] = useState("");
  const [faq, setFaq] = useState("");
  const [chunks, setChunks] = useState("");
  const [note, setNote] = useState("");
  const submit = async () => {
    if (!verdict) return;
    const ids = chunks.split(",").map((s) => s.trim()).filter(Boolean);
    await onSubmit({
      verdict,
      ...(intent.trim() ? { expectedIntent: intent.trim() } : {}),
      ...(faq.trim() ? { expectedFaqId: faq.trim() } : {}),
      ...(ids.length ? { expectedChunkIds: ids } : {}),
      ...(note.trim() ? { note: note.trim() } : {}),
    });
  };
  return (
    <form className="space-y-3 text-sm" onSubmit={(e) => (e.preventDefault(), void submit())}>
      <fieldset className="flex gap-4">
        <legend className="mb-1 font-medium">판정</legend>
        {VERDICTS.map((v) => (
          <label key={v.value} className="flex items-center gap-1">
            <input type="radio" name="verdict" value={v.value} checked={verdict === v.value} onChange={() => setVerdict(v.value)} />
            {v.label}
          </label>
        ))}
      </fieldset>
      <label className="block">정답 의도<Input value={intent} onChange={(e) => setIntent(e.target.value)} aria-label="정답 의도" maxLength={30} /></label>
      <label className="block">정답 FAQ ID<Input value={faq} onChange={(e) => setFaq(e.target.value)} aria-label="정답 FAQ ID" maxLength={100} /></label>
      <label className="block">정답 청크 ID<Input value={chunks} onChange={(e) => setChunks(e.target.value)} aria-label="정답 청크 ID" placeholder="card-002, card-003" /></label>
      <label className="block">메모<Textarea value={note} onChange={(e) => setNote(e.target.value)} aria-label="메모" maxLength={2000} /></label>
      <Button type="submit" disabled={!verdict}>검수 저장</Button>
    </form>
  );
}
```

```tsx
import { useState } from "react";
import { Toaster } from "@/components/ui/sonner";

import { ChatPage } from "@/features/chat/ChatPage";

const PAGES = ["테스트 채팅", "대화 로그", "검수 대기", "지식 데이터", "설정"] as const;
type Page = (typeof PAGES)[number];

export function App() {
  const [token, setTokenState] = useState<string | null>(() => getToken());
  const [page, setPage] = useState<Page>(() => (getToken() ? "테스트 채팅" : "설정"));
  const save = (t: string) => {
    setToken(t);
    setTokenState(getToken());
  };
  const logout = () => {
    clearToken();
    setTokenState(null);
    setPage("설정");
  };
  return (
    <div className="grid h-screen grid-cols-[12rem_1fr]">
      <nav className="space-y-1 border-r p-3">
        {PAGES.map((p) => (
          <button key={p} type="button" onClick={() => setPage(p)} className={`block w-full rounded px-2 py-1 text-left text-sm ${page === p ? "bg-muted font-medium" : "hover:bg-muted"}`}>
            {p}
          </button>
        ))}
      </nav>
      <main className="flex h-screen flex-col overflow-hidden">
        <h1 className="border-b px-4 py-2 text-base font-semibold">{page}</h1>
        <div className="flex-1 overflow-auto">
          {page === "테스트 채팅" && <ChatPage token={token} />}
          {page === "대화 로그" && <LogsPage enabled={!!token} />}
          {page === "검수 대기" && <ReviewPage enabled={!!token} />}
          {page === "지식 데이터" && <KnowledgePage enabled={!!token} />}
          {page === "설정" && <SettingsPage token={token} onSave={save} onLogout={logout} />}
        </div>
      </main>
      <Toaster />
    </div>
  );
}
```
